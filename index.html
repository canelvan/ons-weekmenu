    <script>

        const DB_VERSION = 'v11_stable';

        const DAYS = ['Maandag', 'Dinsdag', 'Woensdag', 'Donderdag', 'Vrijdag', 'Zaterdag', 'Zondag'];

        const CATS = ['Ontbijt', 'Lunch', 'Diner'];


        let currentWeekOffset = 0;

        let dishes = JSON.parse(localStorage.getItem('dishes_' + DB_VERSION)) || [

            { id: 1, name: 'Menemen', category: 'Ontbijt' },

            { id: 2, name: 'Linzensoep', category: 'Lunch' },

            { id: 3, name: 'Tacos', category: 'Diner' }

        ];

        let allWeekPlans = JSON.parse(localStorage.getItem('plans_' + DB_VERSION)) || { "-1": {}, "0": {}, "1": {} };

        let extras = JSON.parse(localStorage.getItem('extras_' + DB_VERSION)) || [];


        function save() {

            localStorage.setItem('dishes_' + DB_VERSION, JSON.stringify(dishes));

            localStorage.setItem('plans_' + DB_VERSION, JSON.stringify(allWeekPlans));

            localStorage.setItem('extras_' + DB_VERSION, JSON.stringify(extras));

        }


        function switchTab(tab) {

            ['planner', 'recipes', 'shopping'].forEach(t => {

                document.getElementById(`section-${t}`).classList.toggle('hidden', t !== tab);

                document.getElementById(`tab-${t}`).classList.toggle('active', t === tab);

            });

            if (tab === 'shopping') renderShoppingList();

        }


        function changeWeek(offset) {

            currentWeekOffset = offset;

            document.querySelectorAll('.week-btn').forEach(btn => btn.classList.remove('active'));

            const btns = document.querySelectorAll('.week-btn');

            if(offset === -1) btns[0].classList.add('active');

            if(offset === 0) btns[1].classList.add('active');

            if(offset === 1) btns[2].classList.add('active');

            updateWeekUI();

        }


        function updateWeekUI() {

            const grid = document.getElementById('weekGrid');

            grid.innerHTML = '';

            const plan = allWeekPlans[currentWeekOffset] || {};


            DAYS.forEach(day => {

                let catHtml = '';

                CATS.forEach(cat => {

                    const saved = plan[`${day}-${cat}`] || '';

                    const options = dishes.filter(d => d.category === cat).map(d => 

                        `<option value="${d.name}" ${saved === d.name ? 'selected' : ''}>${d.name}</option>`

                    ).join('');


                    catHtml += `

                        <div class="p-3 border-b border-orange-100 last:border-0">

                            <span class="text-[10px] uppercase font-black text-orange-300 block mb-1">${cat}</span>

                            <select onchange="savePlan(this.value, '${day}', '${cat}')" class="w-full bg-transparent text-[13px] font-bold text-gray-700 outline-none cursor-pointer">

                                <option value="">-- Kies --</option>

                                ${options}

                                <option value="">(Wissen)</option>

                            </select>

                            <div class="mt-1 text-[11px] font-bold text-primary truncate h-4">${saved}</div>

                        </div>`;

                });


                grid.innerHTML += `

                    <div class="day-card flex flex-col bg-warm-cream border border-orange-100 rounded-2xl shadow-sm overflow-hidden">

                        <div class="bg-primary text-white text-center py-2 text-sm font-bold uppercase tracking-widest">${day}</div>

                        ${catHtml}

                    </div>`;

            });

            const titles = { "-1": "Vorige week", "0": "Deze week", "1": "Volgende week" };

            document.getElementById('currentWeekTitle').innerText = titles[currentWeekOffset];

        }


        window.savePlan = (val, day, cat) => {

            if(!allWeekPlans[currentWeekOffset]) allWeekPlans[currentWeekOffset] = {};

            if(val === "") delete allWeekPlans[currentWeekOffset][`${day}-${cat}`];

            else allWeekPlans[currentWeekOffset][`${day}-${cat}`] = val;

            save();

            updateWeekUI();

        };


        window.addDish = () => {

            const name = document.getElementById('dishName').value.trim();

            const category = document.getElementById('dishCategory').value;

            if(!name) return;

            dishes.push({id: Date.now(), name, category});

            document.getElementById('dishName').value = '';

            save();

            renderFullDishGrid();

            updateWeekUI();

            showToast('Gerecht toegevoegd! ✨');

        };


        function renderFullDishGrid() {

            const grid = document.getElementById('fullDishGrid');

            grid.innerHTML = dishes.map(d => `

                <div class="p-4 bg-white rounded-xl border border-orange-100 flex justify-between items-center shadow-sm">

                    <div><span class="text-[10px] font-black text-orange-300 uppercase">${d.category}</span><h3 class="font-bold text-gray-800">${d.name}</h3></div>

                    <button onclick="deleteDish(${d.id})" class="text-red-200 hover:text-red-500 font-bold text-xl px-2">×</button>

                </div>

            `).join('');

        }


        window.deleteDish = (id) => {

            if(confirm('Gerecht verwijderen uit je boek?')) {

                dishes = dishes.filter(d => d.id !== id);

                save();

                renderFullDishGrid();

                updateWeekUI();

            }

        };


        function renderShoppingList() {

            const plan = allWeekPlans[currentWeekOffset] || {};

            const meals = [...new Set(Object.values(plan))].filter(Boolean);

            

            document.getElementById('autoShoppingList').innerHTML = meals.length ? meals.map(m => `

                <li class="flex items-center gap-3 p-3 bg-warm-cream border border-orange-100 rounded-xl">

                    <input type="checkbox" class="w-5 h-5 accent-[#7d8e7a]"> <span class="font-bold text-gray-700">${m}</span>

                </li>

            `).join('') : '<p class="text-gray-400 italic">Plan maaltijden om deze lijst te vullen.</p>';


            document.getElementById('extraShoppingList').innerHTML = extras.map((ex, i) => `

                <li class="flex items-center justify-between p-3 bg-orange-50 rounded-xl">

                    <div class="flex items-center gap-3"><input type="checkbox" class="w-5 h-5 accent-[#7d8e7a]"> <span class="font-bold text-gray-700">${ex}</span></div>

                    <button onclick="removeExtra(${i})" class="text-red-300 font-bold text-xl px-2">×</button>

                </li>

            `).join('');

        }


        window.addExtraItem = () => {

            const input = document.getElementById('extraItem');

            if(!input.value) return;

            extras.push(input.value);

            input.value = '';

            save();

            renderShoppingList();

        };


        window.removeExtra = (i) => {

            extras.splice(i, 1);

            save();

            renderShoppingList();

        };


        window.clearWeek = () => {

            if(confirm('Hele planning voor deze week wissen?')) {

                allWeekPlans[currentWeekOffset] = {};

                save();

                updateWeekUI();

            }

        };


        function showToast(msg) {

            const t = document.getElementById('toast');

            t.innerText = msg;

            t.classList.remove('translate-y-32');

            setTimeout(() => t.classList.add('translate-y-32'), 2500);

        }


        // Start de app

        window.onload = () => {

            updateWeekUI();

            renderFullDishGrid();

        };
